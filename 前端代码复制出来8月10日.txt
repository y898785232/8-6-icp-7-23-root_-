async function hashPassword(password, email) {
  const data = new TextEncoder().encode(password + email.toLowerCase());
  const hash = await crypto.subtle.digest('SHA-256', data);
  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, '0')).join('');
}

async function init() {
  authClient = await AuthClient.create({ idleOptions: { disableIdle: true } });
  const identity = authClient.getIdentity();
  const agent = new HttpAgent({ identity, host: 'https://icp0.io' });
  backendActor = Actor.createActor(idlFactory, { agent, canisterId: BACKEND_CANISTER_ID });
  if (await authClient.isAuthenticated()) { await handleIILogin(true); }
  else if (sessionId) {
    try {
      const res = await backendActor.getMyProfile(optSession());
      if (res.ok) { currentUser = res.ok; showMain(); }
      else { localStorage.removeItem('sessionId'); sessionId = null; }
    } catch (e) { localStorage.removeItem('sessionId'); sessionId = null; }
  }
}

function disconnectMQTT() {
  if (client) { try { client.end(true); } catch (_) {} client = null; }
  if (offlineCheckTimer) { clearInterval(offlineCheckTimer); offlineCheckTimer = null; }
  mqttConnected = false; lastHeartbeat = 0;
}

function initMQTT() {
  // door_ 璁惧鐢?door/ 涓婚锛孍SP_ 鐢ㄥ師鏉ョ殑 home/esp8266/
  if (isDoorDevice(deviceId)) {
    cmdTopic = 'door/' + deviceId + '/cmd';
    statusTopic = 'door/' + deviceId + '/status';
  } else {
    cmdTopic = 'home/esp8266/cmd/' + deviceId;
    statusTopic = 'home/esp8266/status/' + deviceId;
  }
  disconnectMQTT();
  client = mqtt.connect(MQTT_HOST, {
    clientId: 'web_' + Math.random().toString(16).slice(2, 10),
    username: 'eps8266nn', password: 'yu155414',
    reconnectPeriod: 4000, connectTimeout: 8000, keepalive: 60
  });
  client.on('connect', () => {
    mqttConnected = true;
    setBanner('鉁?浜戠宸茶繛鎺?, '#34c759');
    client.subscribe(statusTopic, { qos: 1 });
    if (isDoorDevice(deviceId)) {
      setTimeout(() => publish('STATUS'), 600);
    } else {
      setTimeout(() => publish('WHO_IS_ONLINE'), 600);
    }
    startOfflineWatch();
  });
  client.on('reconnect', () => setBanner('馃攧 閲嶈繛涓?..', '#ff9500'));
  client.on('close', () => {
    mqttConnected = false;
    setBanner('鈿狅笍 鏂紑', '#ff3b30');
    setButtonsEnabled(false);
    setEl('dev-status', 'badge bg-offline', '宸茬绾?);
  });
  client.on('error', err => setBanner('閿欒: ' + err.message, '#ff3b30'));
  client.on('message', (topic, message) => {
    if (topic !== statusTopic) return;
    const msg = message.toString().trim();
    lastHeartbeat = Date.now();
    const ds = document.getElementById('dev-status');
    if (ds) { ds.className = 'badge bg-online'; ds.textContent = '鍦ㄧ嚎'; }
    setButtonsEnabled(true);
    handleMessage(msg);
  });
}

function handleMessage(msg) {
  const type = getDeviceType(deviceId);

  // 灏濊瘯瑙ｆ瀽 JSON锛堟俯搴︿紶鎰熷櫒锛?  if (msg.startsWith('{')) {
    try {
      const data = JSON.parse(msg);
      if (data.temperature !== undefined && data.temperature !== null) {
        latestTemp = Number(data.temperature);
        const el = document.getElementById('temp-display');
        if (el) el.textContent = latestTemp.toFixed(1) + ' 掳C';
        if (data.sensor_ok === false) {
          setEl('sensor-status', 'badge bg-warning', '寮傚父');
        } else {
          setEl('sensor-status', 'badge bg-online', '姝ｅ父');
        }
        if (data.online === false) {
          setEl('dev-status', 'badge bg-offline', '绂荤嚎');
        }
        return;
      }
    } catch (e) { /* 涓嶆槸鍚堟硶 JSON锛岀户缁寜鏂囨湰澶勭悊 */ }
  }

  if (msg.startsWith('ONLINE_')) { showToast('馃煝 璁惧涓婄嚎'); return; }
  if (msg === 'OTA_START') {
    showToast('馃殌 寮€濮婳TA');
    const b = document.getElementById('btn-ota');
    if (b) { b.textContent = '鍗囩骇涓?..'; b.disabled = true; }
    return;
  }
  if (msg === 'OTA_OK') {
    const b = document.getElementById('btn-ota');
    if (b) { b.textContent = '瑙﹀彂OTA'; b.disabled = false; }
    showToast('鉁?OTA鎴愬姛'); return;
  }
  if (msg === 'OTA_FAIL') {
    const b = document.getElementById('btn-ota');
    if (b) { b.textContent = '瑙﹀彂OTA'; b.disabled = false; }
    showToast('鉂?OTA澶辫触'); return;
  }
  if (msg === 'OTA_NO_NEW') {
    const b = document.getElementById('btn-ota');
    if (b) b.disabled = false;
    showToast('鈩癸笍 宸叉槸鏈€鏂?); return;
  }

  if (type === 'jishe') {
    if (msg === 'ON1') { lampState = true; setEl('lamp-status', 'badge bg-online', '宸插紑'); setBtn('btn-lamp', '鍏崇伅', 'btn btn-danger'); }
    else if (msg === 'OFF1') { lampState = false; setEl('lamp-status', 'badge bg-offline', '宸插叧'); setBtn('btn-lamp', '寮€鐏?, 'btn btn-success'); }
    else if (msg === 'WATER_LOW_ALARM') { setEl('water-status', 'badge bg-warning', '缂烘按'); showToast('鈿狅笍 缂烘按'); }
    else if (msg === 'WATER_NORMAL') { setEl('water-status', 'badge bg-online', '姝ｅ父'); }
    else if (msg === 'DARK') { setEl('light-status', 'badge bg-dark', '鏆?); }
    else if (msg === 'BRIGHT') { setEl('light-status', 'badge bg-bright', '浜?); }
    else if (msg === 'LOCK_TRIGGERED') { showToast('馃敀 闂ㄩ攣宸茶Е鍙?); }
    else if (msg.startsWith('TIMER_INFO_')) {
      const p = msg.split('_');
      if (p.length >= 5) {
        const tc = document.getElementById('timer-enable');
        const st = document.getElementById('start-time');
        const et = document.getElementById('end-time');
        if (tc) tc.checked = p[2] === '1';
        if (st) st.value = p[3];
        if (et) et.value = p[4];
        showToast('瀹氭椂宸插悓姝?);
      }
    }
  }
  if (type === 'shuibeng') {
    if (msg === 'PUMP_ON' || msg === 'ON1') { pumpState = true; setEl('pump-status', 'badge bg-online', '杩愯涓?); }
    else if (msg === 'PUMP_OFF' || msg === 'OFF1') { pumpState = false; setEl('pump-status', 'badge bg-offline', '宸插仠姝?); }
  }
  if (type === 'juanlianmen') {
    if (msg === 'CURTAIN_OPEN') setEl('curtain-status', 'badge bg-online', '宸插紑');
    else if (msg === 'CURTAIN_CLOSE') setEl('curtain-status', 'badge bg-offline', '宸插叧');
    else if (msg === 'CURTAIN_STOP') setEl('curtain-status', 'badge bg-warning', '鍋滄');
  }
  if (type === 'shuangdeng') {
    if (msg === 'ON1') { lampState = true; setEl('lamp-status', 'badge bg-online', '宸插紑'); setBtn('btn-lamp', '鍏崇伅1', 'btn btn-danger'); }
    else if (msg === 'OFF1') { lampState = false; setEl('lamp-status', 'badge bg-offline', '宸插叧'); setBtn('btn-lamp', '寮€鐏?', 'btn btn-success'); }
    else if (msg === 'ON2') { lamp2State = true; setEl('lamp2-status', 'badge bg-online', '宸插紑'); setBtn('btn-lamp2', '鍏崇伅2', 'btn btn-danger'); }
    else if (msg === 'OFF2') { lamp2State = false; setEl('lamp2-status', 'badge bg-offline', '宸插叧'); setBtn('btn-lamp2', '寮€鐏?', 'btn btn-success'); }
    else if (msg === 'PUMP_ON') { pumpState = true; setEl('pump-status', 'badge bg-online', '杩愯涓?); }
    else if (msg === 'PUMP_OFF') { pumpState = false; setEl('pump-status', 'badge bg-offline', '宸插仠姝?); }
  }
}

    showToast(enable ? '鉁?瑙勫垯宸插惎鐢? : '鉁?瑙勫垯宸插仠鐢?);
    loadRules();
  } catch (e) {
    showToast('鉂?' + e.message);
  }
}

async function createRule() {
  const deviceIdVal = document.getElementById('new-rule-device').value.trim();
  const trigger = document.getElementById('new-rule-trigger').value.trim();
  const condition = document.getElementById('new-rule-condition').value.trim();
  const action = document.getElementById('new-rule-action').value.trim();
  const priority = Number(document.getElementById('new-rule-priority').value || 10);
  const enable = document.getElementById('new-rule-enable').checked;
  if (!deviceIdVal || !trigger || !condition || !action) return showToast('璇峰～鍐欏畬鏁磋鍒?);
  try {
    const res = await backendActor.createRule(deviceIdVal, enable, trigger, condition, action, BigInt(priority), optSession());
    if (res.err) return showToast('鉂?' + res.err);
    showToast('鉁?瑙勫垯宸插垱寤?);
    loadRules();
  } catch (e) {
    showToast('鉂?' + e.message);
  }
}

async function deleteRule(id) {
  if (!confirm('纭畾鍒犻櫎璇ヨ鍒欙紵')) return;
  try {
    const res = await backendActor.deleteRule(BigInt(id), optSession());
    if (res.err) return showToast('鉂?' + res.err);
    showToast('鉁?宸插垹闄?);
    loadRules();
  } catch (e) {
    showToast('鉂?' + e.message);
  }
}
